async function invReserveSave() {
  var body = { item_id: document.getElementById('resItem').value, quantity: Number(document.getElementById('resQty').value), warehouse: document.getElementById('resWarehouse').value, work_order: document.getElementById('resRef').value, reason: document.getElementById('resReason').value, request_id: invNewRequestId('web') };
  var btn = document.getElementById('resSave'); if (btn) btn.disabled = true;
  try {
    var r = await fetch('/api/inventory/reserve', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
    var j = await r.json();
    if (!r.ok) { invMsg('resMsg', j.error || 'خطا در ثبت رزرو', false); return; }
    invMsg('resMsg', 'رزرو با موفقیت ثبت شد ✓', true);
    document.getElementById('invReserveForm').classList.add('hidden');
    loadInventory(); renderResSafe();
  } catch (e) { invMsg('resMsg', 'خطا در ارتباط با سرور', false); }
  finally { if (btn) btn.disabled = false; }
}
async function invReserveRelease(id) {
  try {
    var r = await fetch('/api/inventory/reserve/release', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: id, request_id: invNewRequestId('web') }) });

}
